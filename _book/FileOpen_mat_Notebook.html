7.1 Overview

This notebook provides a robust workflow for inspecting MATLAB data files (.mat). It supports both:

  • Classic Files (v4/v5): Unpacked using R.matlab.
  • Modern Files (v7.3+): Unpacked using hdf5r (HDF5 format).

The process reports: 1. Variable Name: The specific object identifier. 2. Class: The data type (e.g., double, cell, H5Group). 3. Dimensions: The size of the array or matrix.

This workflow relies on the R.matlab package.

# FUNCTION: Checks the file's first 4 bytes for the HDF5 signature
# This is a robust way to identify MATLAB v7.3 files.
is_v73_hdf5 <- function(path) {
  # Open file in binary read mode
  con <- file(path, "rb")
  # Ensure connection is closed when the function exits
  on.exit(close(con)) 
  
  # Read the first 4 bytes (HDF5 signature starts with 0x89HDF)
  header <- tryCatch({
    readBin(con, what = "raw", n = 4, size = 1)
  }, error = function(e) {
    # File might be too small or inaccessible
    return(as.raw(0))
  })
  
  # Check if the header matches the HDF5 signature
  return(identical(header, as.raw(c(0x89, 0x48, 0x44, 0x46))))
}

7.4 Generate Metadata Report

MATLAB files come in two broad categories:

  • v4/v5 files, stored in a proprietary binary format

  • v7.3 files, stored as HDF5 (Hierarchical Data Format)

We try to read each .mat file as HDF5 first. If that succeeds, we extract:

  • variable (dataset) names
  • data type (HDF5 class)
  • array dimensions
  • number of dimensions
  • number of elements

If the file is not HDF5, the code falls back to R.matlab::readMat() and extracts the same information from a v4/v5 structure.

# Helper to parse dims from either numeric vector or "3000 x 180 x 180" string
parse_dims <- function(d) {
  # Missing
  if (is.null(d) || (length(d) == 1 && is.na(d))) {
    return(list(str = "-", n_dim = 0L, n_el = NA_integer_))
  }

  # Character like "3000 x 180 x 180"
  if (is.character(d)) {
    parts <- strsplit(d, "x|×")[[1]]
    parts <- trimws(parts)
    parts <- parts[parts != ""]
    nums  <- suppressWarnings(as.integer(parts))

    if (length(nums) == 0 || all(is.na(nums))) {
      # Fallback: just show raw string
      return(list(str = d, n_dim = NA_integer_, n_el = NA_integer_))
    } else {
      return(list(
        str   = paste(nums, collapse = " x "),
        n_dim = length(nums),
        n_el  = as.integer(prod(nums))
      ))
    }
  }

  # Numeric or integer vector
  nums <- suppressWarnings(as.integer(d))
  nums <- nums[!is.na(nums)]
  if (length(nums) == 0) {
    return(list(str = "-", n_dim = 0L, n_el = NA_integer_))
  } else {
    return(list(
      str   = paste(nums, collapse = " x "),
      n_dim = length(nums),
      n_el  = as.integer(prod(nums))
    ))
  }
}

file_summary_list <- list()
var_summary_list  <- list()
idx <- 1L

for (file_path in mat_files) {
  file_name <- basename(file_path)
  cat("\n--- Processing file:", file_name, "---\n")

  file_size_mb <- round(file.info(file_path)$size / (1024^2), 3)

  # 1) TRY: read as HDF5 (MAT v7.3)
  res <- tryCatch({
    f <- hdf5r::H5File$new(file_path, mode = "r")

    # Make sure closing the file never throws a hard error
    on.exit(
      tryCatch(
        f$close_all(),
        error = function(e) {
          message("  close_all() warning for ", file_name, ": ", e$message)
        }
      ),
      add = TRUE
    )

    listing <- f$ls(recursive = TRUE)

    # Keep only datasets
    ds <- listing[listing$obj_type == "H5I_DATASET", , drop = FALSE]

    if (nrow(ds) == 0) stop("HDF5 file contains no datasets")

    dims_list <- ds$dataset.dims

    parsed <- lapply(dims_list, parse_dims)

    dims_str  <- vapply(parsed, function(x) x$str,   character(1))
    num_dims  <- vapply(parsed, function(x) x$n_dim, integer(1))
    num_elems <- vapply(parsed, function(x) x$n_el,  integer(1))

    out <- tibble::tibble(
      FileName      = file_name,
      VariableName  = ds$name,
      Class         = as.character(ds$dataset.type_class),
      Dimensions    = dims_str,
      NumDimensions = num_dims,
      NumElements   = num_elems,
      Format        = "v7.3 (HDF5)"
    )

    cat("  Read as HDF5 (v7.3);", nrow(out), "datasets.\n")
    out
  },
  error = function(e) {
    cat("  HDF5 attempt failed for ", file_name, ": ",
        conditionMessage(e), "\n", sep = "")
    NULL
  })

  # 2) FALLBACK: classic MAT (v4/v5) via readMat, only if HDF5 failed
  if (is.null(res)) {
    res <- tryCatch({
      mat_data  <- R.matlab::readMat(file_path)
      var_names <- names(mat_data)

      if (length(var_names) == 0) {
        cat("  readMat(): file appears empty.\n")
        tibble::tibble(
          FileName      = file_name,
          VariableName  = "(Empty)",
          Class         = "-",
          Dimensions    = "-",
          NumDimensions = 0L,
          NumElements   = NA_integer_,
          Format        = "v4/v5"
        )
      } else {
        cat("  readMat(): ", length(var_names), " variables.\n", sep = "")
        purrr::map_dfr(var_names, function(var) {
          obj  <- mat_data[[var]]
          dims <- dim(obj)
          dims_str <- if (is.null(dims)) {
            paste(length(obj), "(len)")
          } else {
            paste(as.integer(dims), collapse = " x ")
          }
          n_dim <- if (is.null(dims)) 1L else length(dims)
          n_el  <- if (is.null(dims)) length(obj) else as.integer(prod(dims))

          tibble::tibble(
            FileName      = file_name,
            VariableName  = var,
            Class         = class(obj)[1],
            Dimensions    = dims_str,
            NumDimensions = n_dim,
            NumElements   = n_el,
            Format        = "v4/v5"
          )
        })
      }
    },
    error = function(e) {
      cat("  readMat() failed for ", file_name, ": ",
          conditionMessage(e), "\n", sep = "")
      tibble::tibble(
        FileName      = file_name,
        VariableName  = "ERROR",
        Class         = "Error",
        Dimensions    = paste("Failed to read:", conditionMessage(e)),
        NumDimensions = NA_integer_,
        NumElements   = NA_integer_,
        Format        = "Unknown"
      )
    })
  }

  # File-level summary
  file_summary_list[[idx]] <- tibble::tibble(
    FileName     = file_name,
    FullPath     = normalizePath(file_path),
    Format       = unique(res$Format)[1],
    FileSize_MB  = file_size_mb,
    NumVariables = nrow(res)
  )

  # Variable-level summary
  var_summary_list[[idx]] <- res
  idx <- idx + 1L
}

7.5 Notes and limitations

MATLAB .mat files can vary significantly in format and complexity. We have some limitations to keep in mind:

  • Very large datasets may not be fully loaded and may be summarized based on sampling.
  • Deeply nested MATLAB structs or objects may require custom handling.
  • Character encoding in .mat files may differ across MATLAB versions.
  • Complex cell arrays can contain heterogeneous types that cannot be summarized numerically.
